# 5) Cloudinary orphan cleanup helper (optional)

Node script to detect Cloudinary images referenced in DB vs Cloudinary account and optionally delete unreferenced images:

```js
// scripts/cloudinary-orphan-cleanup.js
const cloudinary = require('cloudinary').v2;
const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();
cloudinary.config({ cloud_name: process.env.CLOUDINARY_CLOUD_NAME, api_key: process.env.CLOUDINARY_API_KEY, api_secret: process.env.CLOUDINARY_API_SECRET });

async function run() {
  const dbResources = (await prisma.gallery.findMany({ select: { publicId: true } })).map(r => r.publicId);
  // list Cloudinary folder resources
  let nextCursor = null;
  const cloudResources = [];
  do {
    const res = await cloudinary.api.resources({ type: 'upload', prefix: 'smarthotel', max_results: 500, next_cursor: nextCursor });
    res.resources.forEach(r => cloudResources.push(r.public_id));
    nextCursor = res.next_cursor;
  } while (nextCursor);

  const orphans = cloudResources.filter(id => !dbResources.includes(id));
  console.log('Cloudinary orphan count:', orphans.length);
  // Do NOT delete automatically; print list for review
  orphans.slice(0, 200).forEach(o => console.log(o));
  process.exit(0);
}
run();
```

```yaml
name: full-validate
on:
  workflow_dispatch:
jobs:
  validate:
    runs-on: ubuntu-latest
    env:
      API_BASE_URL: ${{ secrets.API_BASE_URL }}
      DATABASE_URL: ${{ secrets.DATABASE_URL }}
      CLOUDINARY_CLOUD_NAME: ${{ secrets.CLOUDINARY_CLOUD_NAME }}

      CLOUDINARY_API_SECRET: ${{ secrets.CLOUDINARY_API_SECRET }}
      ADMIN_API_TOKEN: ${{ secrets.ADMIN_API_TOKEN }}
      ADMIN_EMAIL: ${{ secrets.ADMIN_EMAIL }}
      ADMIN_PASSWORD: ${{ secrets.ADMIN_PASSWORD }}
    steps:
      - uses: actions/checkout@v4
      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: '20'
      - run: npm ci
      - name: Run DB Integrity
        run: node scripts/db-integrity.js
      - name: Run Cloudinary check
        run: node scripts/cloudinary-check.js
      - name: Run API tests
        run: npx jest --runInBand
      - name: Run Playwright
        uses: microsoft/playwright-github-action@v1
        with:
          run: npx playwright test --project=chromium
      - name: Upload artifacts
        uses: actions/upload-artifact@v4
        with:
          name: validation-artifacts
          path: artifacts/
```
